## Critical Security Pattern: Access Token in Memory (Pitfall 1)

Access token is stored in AuthContext state ONLY, never in localStorage or sessionStorage. On page refresh, state is lost and user is logged out. This is acceptable UX — user can use refresh token cookie to silently re-authenticate (Wave 4 POST /api/auth/refresh).

Refresh token is stored in HttpOnly cookie by backend; frontend never touches it. Browser automatically sends it with credentials.

## Refresh Semaphore Pattern (Pitfall 2)

When axios response interceptor detects 401:
1. Check `isRefreshing` flag
2. If false: set flag to true, call refresh endpoint, collect all queued requests
3. If true: queue this request and await refresh result
4. When refresh completes: drain queue, send all queued requests with new token
5. If refresh fails (401): log out all users

This prevents thundering herd when multiple requests fail simultaneously.

## Axios Configuration

```typescript
api.interceptors.request.use((config) => {
  if (token) config.headers.Authorization = `Bearer ${token}`
  return config
})

api.interceptors.response.use(
  (response) => response,
  async (error) => {
    if (error.response?.status === 401) {
      // Handle 401
    }
  }
)
```

  <files>frontend/src/services/api.ts</files>
  <action>
    Update `frontend/src/services/api.ts` to add request interceptor (Bearer token), response interceptor (401 handling with refresh semaphore), and refresh token functions.
    
    **Requirements:**
    - Create request interceptor to add `Authorization: Bearer <token>` header if token exists
    - Create response interceptor to handle 401 errors with token refresh
    - Implement refresh semaphore: queue requests while refresh is in flight
    - Export api instance and refresh functions
    
    **File structure:**
    ```typescript
    import axios, { AxiosError, InternalAxiosRequestConfig } from 'axios'
    import type { AuthResponse } from '../types/auth.js'
    
    const API_URL = import.meta.env.VITE_API_URL || 'http://localhost:5000'
    
    // Create axios instance
    export const api = axios.create({
      baseURL: `${API_URL}/api`,
      timeout: 60000,
      withCredentials: true, // IMPORTANT: Include cookies (refresh token)
    })
    
    // Refresh token semaphore
    let isRefreshing = false
    let refreshSubscribers: ((token: string) => void)[] = []
    
    function subscribeTokenRefresh(callback: (token: string) => void) {
      refreshSubscribers.push(callback)
    }
    
    function onRefreshed(token: string) {
      refreshSubscribers.forEach((callback) => callback(token))
      refreshSubscribers = []
    }
    
    // Request interceptor: add Authorization header
    api.interceptors.request.use((config: InternalAxiosRequestConfig) => {
      const token = sessionStorage.getItem('access_token') // Get from wherever AuthContext stores it
      // Actually, we can't access React state from here. We'll need to inject the token from AuthContext
      // Instead: let AuthContext set a module-level variable or use a getter
      return config
    })
    
    // Response interceptor: handle 401 with refresh
    api.interceptors.response.use(
      (response) => response,
      async (error: AxiosError) => {
        const config = error.config as InternalAxiosRequestConfig
        
        if (error.response?.status === 401 && config && !config._retry) {
          config._retry = true
          
          if (!isRefreshing) {
            isRefreshing = true
            
            try {
              const { data } = await api.post<{ accessToken: string }>('/auth/refresh', {}, {
                _retry: true, // Don't retry this refresh call itself
              } as any)
              
              isRefreshing = false
              onRefreshed(data.accessToken)
              
              // Retry original request with new token
              return api(config)
            } catch (refreshError) {
              isRefreshing = false
              refreshSubscribers = []
              // Logout user
              throw refreshError
            }
          }
          
          // Queue this request while refresh is in flight
          return new Promise((resolve) => {
            subscribeTokenRefresh((token: string) => {
              // Will be called with new token when refresh completes
              // Retry original request
              resolve(api(config))
            })
          })
        }
        
        return Promise.reject(error)
      }
    )
    
    // Export auth endpoints
    export async function loginUser(email: string, password: string) {
      const response = await api.post<AuthResponse>('/auth/login', { email, password })
      return response.data
    }
    
    export async function signupUser(email: string, password: string) {
      const response = await api.post<AuthResponse>('/auth/signup', { email, password })
      return response.data
    }
    
    export async function logoutUser() {
      await api.post('/auth/logout')
    }
    
    export async function refreshAccessToken() {
      try {
        const response = await api.post<{ accessToken: string }>('/auth/refresh', {}, {
          _retry: true,
        } as any)
        return response.data.accessToken
      } catch (error) {
        return null
      }
    }
    
    export async function getCurrentUser() {
      const response = await api.get('/auth/me')
      return response.data
    }
    
    // Legacy upload functions
    export async function uploadGardenImage(file: File) {
      const formData = new FormData()
      formData.append('image', file)
      const response = await api.post('/upload', formData, {
        headers: { 'Content-Type': 'multipart/form-data' },
      })
      return response.data
    }
    
    export function getImageUrl(imageId: string): string {
      return `${API_URL}/api/image/${imageId}`
    }
    ```
    
    **Key patterns:**
    - withCredentials: true = send refresh token cookie with all requests
    - isRefreshing flag prevents concurrent refresh calls
    - refreshSubscribers array queues requests waiting for refresh
    - Response interceptor checks error.response?.status === 401
    - Retry logic: config._retry flag prevents infinite loops
    - onRefreshed drains queue and retries all queued requests
    
    **Problem: We need to inject the token from AuthContext into the request interceptor, but we can't access React state from an interceptor. Solution:**
    - AuthContext will call api.setToken(token) when token changes
    - We'll add a setToken function to manage this
    
    **Revised approach:**
    Add a setToken function that the AuthContext calls:
    ```typescript
    let currentAccessToken: string | null = null
    
    export function setAccessToken(token: string | null) {
      currentAccessToken = token
    }
    
    api.interceptors.request.use((config) => {
      if (currentAccessToken) {
        config.headers.Authorization = `Bearer ${currentAccessToken}`
      }
      return config
    })
    ```

  <action>
    Create `frontend/src/contexts/AuthContext.tsx` with React Context for global auth state, provider component, and internal hook.
    
    **Requirements:**
    - AuthContext: stores user, isAuthenticated, isLoading, login, signup, logout, refreshAccessToken
    - AuthProvider: component wrapping app, manages state and calls api functions
    - useAuthContext: internal hook to access context (with error checking)
    - State stored in useState: user, isLoading, accessToken
    - accessToken never stored in localStorage (only in state)
    - login(email, password): calls api.loginUser, stores token in state, sets user
    - signup(email, password): calls api.signupUser, stores token, sets user
    - logout(): calls api.logoutUser, clears token and user state
    - refreshAccessToken(): calls api.refreshAccessToken, updates token in state
    - All async operations set isLoading during request
    
    **File format:**
    ```typescript
    import React, { createContext, useState, useCallback } from 'react'
    import type { AuthContextType, User } from '../types/auth.js'
    import * as authApi from '../services/api.js'
    
    export const AuthContext = createContext<AuthContextType | undefined>(undefined)
    
    export function AuthProvider({ children }: { children: React.ReactNode }) {
      const [user, setUser] = useState<User | null>(null)
      const [accessToken, setAccessToken] = useState<string | null>(null)
      const [isLoading, setIsLoading] = useState(false)
      
      // Sync accessToken to api service
      React.useEffect(() => {
        authApi.setAccessToken(accessToken)
      }, [accessToken])
      
      const login = useCallback(async (email: string, password: string) => {
        setIsLoading(true)
        try {
          const data = await authApi.loginUser(email, password)
          setAccessToken(data.accessToken)
          setUser(data.user)
        } finally {
          setIsLoading(false)
        }
      }, [])
      
      const signup = useCallback(async (email: string, password: string) => {
        setIsLoading(true)
        try {
          const data = await authApi.signupUser(email, password)
          setAccessToken(data.accessToken)
          setUser(data.user)
        } finally {
          setIsLoading(false)
        }
      }, [])
      
      const logout = useCallback(async () => {
        setIsLoading(true)
        try {
          await authApi.logoutUser()
        } finally {
          setAccessToken(null)
          setUser(null)
          setIsLoading(false)
        }
      }, [])
      
      const refreshAccessToken = useCallback(async () => {
        try {
          const newToken = await authApi.refreshAccessToken()
          if (newToken) {
            setAccessToken(newToken)
            return newToken
          } else {
            // Refresh failed, log out
            setAccessToken(null)
            setUser(null)
            return null
          }
        } catch (error) {
          setAccessToken(null)
          setUser(null)
          return null
        }
      }, [])
      
      const value: AuthContextType = {
        user,
        isAuthenticated: !!accessToken && !!user,
        isLoading,
        login,
        signup,
        logout,
        refreshAccessToken,
      }
      
      return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>
    }
    
    export function useAuthContext(): AuthContextType {
      const context = React.useContext(AuthContext)
      if (!context) {
        throw new Error('useAuthContext must be used within AuthProvider')
      }
      return context
    }
    ```
    
    **Why this structure:**
    - accessToken stored in state only (not localStorage)
    - useEffect syncs token to api service when it changes
    - isLoading tracks async operations
    - login/signup set both token and user
    - logout clears both
    - refreshAccessToken handles failed refresh (logs out)
    - useAuthContext throws error if used outside provider (catches config bugs early)
